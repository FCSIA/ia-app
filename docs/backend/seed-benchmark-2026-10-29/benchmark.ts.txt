import { v } from "convex/values";
import { hashPassword } from "better-auth/crypto";
import { components, internal } from "./_generated/api";
import { internalAction, internalMutation, internalQuery } from "./_generated/server";
import type { Id } from "./_generated/dataModel";
import { createLinkedUser } from "./users/_lib";
import { incrementUserCounts } from "./stats";
import { LOCAL_DEMO_PASSWORD } from "./seed";

function guard() {
  if (process.env.ALLOW_LOCAL_DEMO_SEED !== "true") throw new Error("Local demo only");
}

export const snapshot = internalQuery({
  args: {}, returns: v.object({ users: v.number(), benchmarkUsers: v.number(), trackedUsers: v.number() }),
  handler: async (ctx) => {
    const users = await ctx.db.query("users").withIndex("by_uniqueId").take(1000);
    const stats = await ctx.db.query("appStats").withIndex("by_key", (q) => q.eq("key", "user-counts")).unique();
    return { users: users.length, benchmarkUsers: users.filter((user) => user.uniqueId.startsWith("SEEDBENCH")).length, trackedUsers: stats?.users ?? 0 };
  },
});

export const noop = internalMutation({ args: {}, returns: v.null(), handler: async () => { guard(); return null; } });
export const hash = internalMutation({ args: {}, returns: v.null(), handler: async () => { guard(); await hashPassword(LOCAL_DEMO_PASSWORD); return null; } });

export const create = internalMutation({
  args: { uniqueId: v.string() }, returns: v.id("users"),
  handler: async (ctx, { uniqueId }) => {
    guard();
    if (!uniqueId.startsWith("SEEDBENCH")) throw new Error("Invalid benchmark ID");
    const { user } = await createLinkedUser(ctx, {
      uniqueId, firstName: "Seed", lastName: "Benchmark", roles: ["STUDENT"],
      email: `${uniqueId.toLowerCase()}@example.invalid`, password: LOCAL_DEMO_PASSWORD, mustResetPassword: false,
    });
    return user._id;
  },
});

export const remove = internalMutation({
  args: { id: v.id("users") }, returns: v.null(),
  handler: async (ctx, { id }) => {
    guard();
    const user = await ctx.db.get(id);
    if (!user) return null;
    if (!user.uniqueId.startsWith("SEEDBENCH") || !user.email?.endsWith("@example.invalid")) throw new Error("Refusing non-benchmark user");
    if (user.authUserId) {
      const maintenanceSecret = process.env.BETTER_AUTH_MAINTENANCE_SECRET;
      if (!maintenanceSecret) throw new Error("Missing maintenance secret");
      await ctx.runMutation(components.betterAuth.dev.deleteUserData, { userId: user.authUserId, maintenanceSecret });
    }
    await ctx.db.delete(id);
    await incrementUserCounts(ctx, user, -1);
    return null;
  },
});

export const scheduled = internalMutation({
  args: { id: v.id("seedRuns"), remaining: v.number() }, returns: v.null(),
  handler: async (ctx, { id, remaining }) => {
    guard();
    const now = Date.now();
    await ctx.db.patch(id, { cursor: String(remaining), updatedAt: now, ...(remaining === 1 ? { status: "COMPLETE" as const, completedAt: now } : {}) });
    if (remaining > 1) await ctx.scheduler.runAfter(0, internal.seedBenchmark.scheduled, { id, remaining: remaining - 1 });
    return null;
  },
});

export const queue = internalMutation({
  args: { count: v.number() }, returns: v.id("seedRuns"),
  handler: async (ctx, { count }) => {
    guard();
    const now = Date.now();
    const id = await ctx.db.insert("seedRuns", { key: "seed-benchmark", runId: String(now), phase: "benchmark", status: "RUNNING", startedAt: now, updatedAt: now });
    await ctx.scheduler.runAfter(0, internal.seedBenchmark.scheduled, { id, remaining: count });
    return id;
  },
});
export const status = internalQuery({
  args: { id: v.id("seedRuns") }, returns: v.union(v.null(), v.object({ startedAt: v.number(), completedAt: v.union(v.number(), v.null()) })),
  handler: async (ctx, { id }) => { const run = await ctx.db.get(id); return run ? { startedAt: run.startedAt, completedAt: run.completedAt ?? null } : null; },
});
export const removeRun = internalMutation({
  args: { id: v.id("seedRuns") }, returns: v.null(),
  handler: async (ctx, { id }) => { guard(); const run = await ctx.db.get(id); if (run?.key === "seed-benchmark") await ctx.db.delete(id); return null; },
});

export const run = internalAction({
  args: {}, returns: v.object({ noopMs: v.array(v.number()), hashMs: v.array(v.number()), createMs: v.array(v.number()), schedulerHops: v.number(), schedulerMs: v.number(), cleanedUsers: v.number() }),
  handler: async (ctx): Promise<{ noopMs: number[]; hashMs: number[]; createMs: number[]; schedulerHops: number; schedulerMs: number; cleanedUsers: number }> => {
    guard();
    const noopMs: number[] = [], hashMs: number[] = [], createMs: number[] = [];
    const ids: Id<"users">[] = [];
    let runId: Id<"seedRuns"> | null = null;
    const prefix = `SEEDBENCH${Date.now()}`;
    try {
      for (let i = 0; i < 14; i++) {
        let start = Date.now();
        await ctx.runMutation(internal.seedBenchmark.noop, {});
        if (i >= 2) noopMs.push(Date.now() - start);
        start = Date.now();
        await ctx.runMutation(internal.seedBenchmark.hash, {});
        if (i >= 2) hashMs.push(Date.now() - start);
        start = Date.now();
        ids.push(await ctx.runMutation(internal.seedBenchmark.create, { uniqueId: `${prefix}${i}` }));
        if (i >= 2) createMs.push(Date.now() - start);
      }
      const schedulerHops = 25;
      runId = await ctx.runMutation(internal.seedBenchmark.queue, { count: schedulerHops });
      let schedulerMs = -1;
      for (let i = 0; i < 200; i++) {
        const run = await ctx.runQuery(internal.seedBenchmark.status, { id: runId });
        if (run?.completedAt != null) { schedulerMs = run.completedAt - run.startedAt; break; }
        await new Promise((resolve) => setTimeout(resolve, 50));
      }
      if (schedulerMs < 0) throw new Error("Scheduled chain timed out");
      return { noopMs, hashMs, createMs, schedulerHops, schedulerMs, cleanedUsers: ids.length };
    } finally {
      for (const id of ids) await ctx.runMutation(internal.seedBenchmark.remove, { id });
      if (runId) await ctx.runMutation(internal.seedBenchmark.removeRun, { id: runId });
    }
  },
});
